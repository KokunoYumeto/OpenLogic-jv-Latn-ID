\olsection{Kasahihan}

\begin{explain}
Sistem !!^a{derivation}, kayata deduksi aksiomatik, iku \emph{sah} yen
ora bisa !!{derive} bab kang sejatine ora lumaku. Mula kasahihan iku
sawijining sipat kaslametan kang dijamin kanggo sistem
!!{derivation}. Gumantung marang sipat ing teori bukti kang dirembug,
kita kepengin ngerti, upamane, yen
\begin{enumerate}
\item saben teorema $!A$ kang derivabel tanpa premis iku \iftag{FOL}{valid}{tautologi};
\item yen $!A$ iku !!{derivable} saka sawetara formula liyane~$\Gamma$,
  formula iku uga dadi akibate;
\item yen sawijining himpunan !!{formula}s~$\Gamma$ ora konsisten,
  himpunan iku ora bisa dicukupi.
\end{enumerate}
Iki sipat-sipat wigati saka sistem !!a{derivation}. Yen salah sijine
ora lumaku, sistem !!{derivation} kasebut cacad---sistem iku bakal
!!{derive} kakehan. Mula, netepake kasahihan sistem
!!a{derivation} iku wigati banget.
\end{explain}

\begin{prop}
  Yen $!A$ minangka aksioma, mula
  \iftag{FOL}
    {$\Sat{M}{!A}[s]$ kanggo saben !!{structure}~$\Struct{M}$ lan penetapan~$s$.}
    {$\pSat{v}{!A}$ kanggo saben !!{valuation}~$\pAssign{v}$.}
\end{prop}

\begin{proof}
\iftag{FOL}{Saben skema proposisional nduwèni nilai bener kanggo
  saben kombinasi nilai kabeneran metavariabelé, kaya kang bisa
  dipriksa nganggo tabel kabeneran.
  Kanggo \olref[qua]{ax:q1}, upamakna $t$ term katutup lan
  $\Sat{M}{\lforall[x][!A]}[s]$. Pilih $s'$ kang cocog karo $s$
  ing njaba $x$, kanthi $s'(x)=\Value{t}{M}[s]$. Definisi panyukupan
  mènèhi $\Sat{M}{!A}[s']$, lan
  \olref[syn][ext]{prop:ext-formulas} mènèhi
  $\Sat{M}{\Subst{!A}{t}{x}}[s]$.
  Kanggo \olref[qua]{ax:q2}, upamakna $\Sat{M}{\Subst{!A}{t}{x}}[s]$.
  Lema substitusi mènèhi $\Sat{M}{!A}[s']$ kanggo pangaji kang padha
  mau. Pangaji iki dadi saksi kanggo $\Sat{M}{\lexists[x][!A]}[s]$.
  Mula loro skema kuantor mau valid.
}{Gawea tabel kabeneran kanggo saben aksioma supaya bisa dipriksa
  yèn kabèh iku tautologi.}
\end{proof}

\begin{thm}[Kasahihan]
\ollabel{thm:soundness}
\iftag{FOL}{Kanggo formula tataran kapisan, $\Gamma \Entails !A$ tegesé kanggo
saben struktur $\Struct{M}$ lan pangaji $s$, yèn
$\Sat{M}{!B}[s]$ kanggo saben $!B\in\Gamma$, mula $\Sat{M}{!A}[s]$.
Kanggo ukara, iki padha karo akibat semantis kang wis ditetepaké.
}{}
Yèn $\Gamma \Proves !A$, mula $\Gamma \Entails !A$.
\end{thm}

\begin{proof}
Kanthi induksi miturut cacah larik ing !!{derivation} kang formula
pungkasané $!A$ saka $\Gamma$. Yèn larik pungkasan iku instansi
aksioma, larik iku \iftag{FOL}{valid}{tautologi}, mula
$\Gamma \Entails !A$. Yèn $!A\in\Gamma$, saben
\iftag{FOL}{struktur lan pangaji}{valuasi} kang nyukupi premis uga
nyukupi $!A$, mula $\Gamma \Entails !A$.
\iftag{FOL}{Yèn derivasi nganggo konstanta pambantu, jembarna
struktur nganggo interpretasi sembarang kanggo konstanta mau.
Panyukupan formula ing basa wiwitan tetep padha.}{}

Yèn larik pungkasan saka !!{derivation} yaiku asil modus ponens,
ana larik sadurungé awujud $!B$ lan $!B\lif !A$. Miturut hipotesis
induksi, $\Gamma\Entails !B$ lan $\Gamma\Entails !B\lif !A$.
Mula $\Gamma\Entails !A$, amarga kabeneran $!B$ lan $!B\lif !A$
mènèhi kabeneran $!A$ \iftag{FOL}{ing struktur lan pangaji kang padha}{ing valuasi kang padha}.

\iftag{FOL}{Yèn larik pungkasan nganggo \QR{} universal, wujudé
$!C\lif\lforall[x][!B(x)]$, lan ing larik sadurungé
ana $!C\lif !B(c)$. Konstanta $c$ absèn saka $\Gamma$, $!C$, lan
matriks $!B(x)$. Miturut induksi, $\Gamma\Entails !C\lif !B(c)$.
Pilih $\Struct{M}$ lan pangaji $s$ kang nyukupi $\Gamma$, lan
upamakna $\Sat{M}{!C}[s]$. Kanggo saben unsur $d$ ing domain,
tetepna $s'$ padha karo $s$ kajaba $s'(x)=d$.

Tetepna $\Struct{M'}$ padha karo $\Struct{M}$ kajaba $\Assign{c}{M'}=d$.
Amarga $c$ ora muncul ing $\Gamma$ lan $!C$,
\olref[syn][ext]{prop:extensionality} njamin formula mau tetep
dicukupi ing $\Struct{M'}$ kanthi pangaji $s$. Hipotesis induksi
mènèhi $\Sat{M'}{!B(c)}[s]$. Miturut \olref[syn][ext]{prop:ext-formulas},
iki padha karo $\Sat{M'}{!B(x)}[s']$. Amarga matriks uga ora ngemot
$c$, kita nduwèni $\Sat{M}{!B(x)}[s']$. Iki lumaku kanggo saben $d$,
mula $\Sat{M}{\lforall[x][!B(x)]}[s]$ miturut
\olref[syn][ass]{prop:sat-quant}. Dadi
$\Gamma\Entails !C\lif\lforall[x][!B(x)]$.

Kasus nalika $!A$ nduweni dhasar \QR{} nanging awujud
$\lexists[x][!B(x)] \lif !C$ ditinggalake minangka gladhen.}{}
\end{proof}

\tagprob{FOL}
\begin{prob}
  Rampungna bukti \olref[fol][axd][sou]{thm:soundness}.
\end{prob}
\tagendprob

\begin{cor}
\ollabel{cor:weak-soundness}
Yen $\Proves !A$, mula $!A$ iku \iftag{FOL}{valid}{tautologi}.
\end{cor}

\begin{cor}
\ollabel{cor:consistency-soundness}
Yen $\Gamma$ bisa dicukupi, mula himpunan kasebut konsisten.
\end{cor}

\begin{proof}
Kita mbuktekaké kontraposisi. Yèn $\Gamma$ ora konsisten, mula
$\Gamma\Proves\lfalse$. Miturut \olref{thm:soundness},
$\Gamma\Entails\lfalse$. Amarga $\lfalse$ tansah salah,
\iftag{FOL}{ora ana pasangan struktur lan pangaji kang nyukupi $\Gamma$}{ora ana valuasi kang nyukupi $\Gamma$}.
Dadi $\Gamma$ ora bisa dicukupi.
\end{proof}
