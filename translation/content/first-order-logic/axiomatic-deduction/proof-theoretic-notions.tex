\olsection{Gagasan Teoretis-Bukti}

\begin{explain}
Kaya nalika kita nemtokake sawetara gagasan semantis wigati
(\iftag{FOL}{validitas}{tautologi}, akibat, lan bisa dicukupi), saiki
kita nemtokake \emph{gagasan teoretis-bukti} kang cocog. Gagasan iki
ora ditemtokake kanthi nggunakake panyukupan !!{formula}s \iftag{FOL}{ing struktur kanthi pangaji variabel}{ing valuasi}, nanging kanthi nggunakake !!{derivability} utawa
!!{nonderivability} formula-formula tartamtu. Panemuan wigati yaiku
manawa gagasan kasebut cocog siji lan sijine. Kecocogan kasebut dadi isi
\emph{teorema kasahihan} lan \emph{teorema kelengkapan}.
\end{explain}

\begin{defn}[!!^{derivability}]
!!^a{formula}~$!A$ iku \emph{!!{derivable}} saka $\Gamma$, ditulis
$\Gamma \Proves !A$, yen ana !!a{derivation} saka~$\Gamma$ kang
formula pungkasane~$!A$.
\end{defn}

\begin{defn}[Teorema]
!!^a{formula}~$!A$ iku \emph{teorema} yen $!A$ dadi formula pungkasan
ing !!a{derivation} saka himpunan kosong. Kita nulis $\Proves !A$ yen $!A$
iku teorema, lan $\Proves/ !A$ yen dudu teorema.
\end{defn}

\begin{defn}[Konsistensi]
Himpunan !!{formula}s $\Gamma$ iku \emph{konsisten} yen lan mung yen
$\Gamma\Proves/ \lfalse$; yen ora, himpunan iku \emph{ora konsisten}.
\end{defn}

\begin{prop}[Refleksivitas]
\ollabel{prop:reflexivity}
Yen $!A \in \Gamma$, mula $\Gamma \Proves !A$.
\end{prop}

\begin{proof}
  !!{formula}~$!A$ dhewe yaiku !!a{derivation} kang formula pungkasane
  $!A$ lan kang dijupuk saka~$\Gamma$.
\end{proof}

\begin{prop}[Monotonisitas]
\ollabel{prop:monotonicity}
Yen $\Gamma \subseteq \Delta$ lan $\Gamma \Proves !A$, mula $\Delta
\Proves !A$.
\end{prop}

\begin{proof}
Induksi miturut cacah larik ing derivasi saka konteks wiwitan.
Saben premis saka $\Gamma$ uga premis saka $\Delta$.
Yèn larik pungkasan iku instansi aksioma, tetepna larik iku.
Kanggo modus ponens, gunakna hipotesis induksi marang loro prefiks
kang pungkasané formula premis inferensi, banjur gabungna derivasi
saka $\Delta$ lan trapna \MP{}.
\iftag{FOL}{%
Kanggo aturan kuantor, jenengna eigenkonstanta ing inferensi
pungkasan $a$. Yèn $a$ ana ing $\Delta$, gantenana $a$ kanthi seragam ing prefiks premisé nganggo $b$ kang anyar.
Pilih $b$ kang ora ana ing $\Delta$, prefiks lawas lan kesimpulan.
Panggantos iki njaga aksioma lan kabeh aturan ing prefiks, amarga
$a$ absèn saka $\Gamma$ lan $b$ durung tau dienggo ing kono.
Miturut hipotesis induksi, premis kang wis diganti nduwèni derivasi
saka $\Delta$. Trapna aturan kuantor nganggo $b$. Kesimpulan tetep
padha amarga $a$ absèn saka kesimpulan aturan iku. Yèn $a$ wis absèn
saka $\Delta$, hipotesis induksi ngidini kita tetep nganggo $a$.
Derivasi bisa nganggo tambahan winates konstanta pambantu kang
dipilih saka jeneng anyar ing njaba basa wiwitan, kanthi premis lan
kesimpulan pungkasan tetep ing basa wiwitan.}{}%
\end{proof}

\begin{prop}[Transitivitas]
\ollabel{prop:transitivity}
Yen $\Gamma \Proves !A$ lan $\{!A\} \cup \Delta \Proves
!B$, mula $\Gamma \cup \Delta \Proves !B$.
\end{prop}

\begin{proof}
  Upamana $\{!A\}\cup\Delta\Proves !B$. Kita nganggo induksi miturut
  cacah larik ing derivasi kang pungkasané $!B$ saka konteks iki.
  Yèn larik $!B_i = !A$ dadi premis, anggo derivasi saka $\Gamma$ lan
  Monotonisitas kanggo ngasilaké $!A$ saka $\Gamma\cup\Delta$.
  Premis saka $\Delta$ tetep premis, lan instansi aksioma tetep sah.
  Kanggo langkah \MP{}, trapna induksi marang loro derivasi kang
  pungkasané premis inferensi. Gabungna asilé lan trapna \MP{}.
  \iftag{FOL}{%
  Kanggo aturan kuantor, yèn eigenkonstanta $a$ ora absèn saka
  $\Gamma\cup\Delta$, pilih konstanta anyar kaya ing bukti
  Monotonisitas, lan pesthèkna uga konstanta iku absèn saka $!A$.
  Ing prefiks kang pungkasané premis aturan kuantor, ganti $a$ nganggo
  konstanta anyar iku. Hipotesis aturan njamin
  $a$ absèn saka $\{!A\}\cup\Delta$ lan saka kesimpulan inferensi.
  Ing kasus iki, trapna induksi marang derivasi premis kang wis diowahi.
  Trapna aturan kuantor marang premis kang wis diasilaké saka konteks
  gabungan. Yèn $a$ wis absèn saka konteks gabungan, trapna induksi
  langsung marang derivasi premis lawas lan nganggo aturan kang padha.}{}%
  Mula $\Gamma\cup\Delta\Proves !B$.
\end{proof}

Gatekna yen iki mligine ateges: yen $\Gamma \Proves !A$ lan $!A
\Proves !B$, mula $\Gamma \Proves !B$. Uga ndherek yen $!A_1,
\dots, !A_n \Proves !B$ lan $\Gamma \Proves !A_i$ kanggo saben~$i$,
mula $\Gamma \Proves !B$.

\begin{prop}
\ollabel{prop:incons}
$\Gamma$ ora konsisten yen lan mung yen $\Gamma \Proves !A$ kanggo
saben~$!A$.
\end{prop}

\begin{proof}
Gladhen.
\end{proof}

\begin{prop}[Kekompakan]
  \ollabel{prop:proves-compact}
  \begin{enumerate}
  \item Yen $\Gamma \Proves !A$, mula ana subhimpunan winates
    $\Gamma_0 \subseteq \Gamma$ saengga $\Gamma_0 \Proves !A$.
  \item Yen saben subhimpunan winates saka~$\Gamma$ iku konsisten,
    mula $\Gamma$ konsisten.
  \end{enumerate}
\end{prop}

\begin{proof}
  \begin{enumerate}
    \item Yen $\Gamma \Proves !A$, mula ana rerangken winates
      !!{formula}s $!A_1$, \dots,~$!A_n$ saengga $!A \ident !A_n$
      lan saben $!A_i$ iku aksioma logis, !!a{element} saka~$\Gamma$,
      utawa ndherek saka !!{formula}s sadurunge miturut salah siji
      aturan inferensi kang diidini. Tetepna $\Gamma_0$ minangka
      himpunan kabeh $!A_i$ kang kalebu ing~$\Gamma$. Banjur
      !!{derivation} kasebut uga dadi !!a{derivation} saka~$\Gamma_0$,
      mula $\Gamma_0
      \Proves !A$. Syarat konstanta anyar tetep lumaku nalika konteks disuda saka
      $\Gamma$ dadi $\Gamma_0$, mula langkah-langkah aturan kuantor uga sah.
    \item Iki kontraposisi saka~(1) kanggo kasus mligi $!A
      \ident \lfalse$.
\end{enumerate}
\end{proof}
